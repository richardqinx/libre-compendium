\begin{tikzpicture}[font=\small]
  % Subspace span(X) as a parallelogram
  \coordinate (O) at (0,0);
  \coordinate (P1) at (-1.6,-1.1);
  \coordinate (P2) at (5.2,-1.1);
  \coordinate (P3) at (7.0,1.3);
  \coordinate (P4) at (0.2,1.3);

  \fill[lc-def!8] (P1) -- (P2) -- (P3) -- (P4) -- cycle;
  \draw[draw=lc-def!60, line width=0.7pt] (P1) -- (P2) -- (P3) -- (P4) -- cycle;
  \node[lc label, text=lc-def] at (5.3,-0.75) {$\mathrm{span}(\bm{X})\subset\R^n$};

  % Basis columns of X
  \draw[lc aux, -{Stealth}] (O) -- (2.2,-0.55) node[below right, lc label] {$\bm{X}_{:,1}$};
  \draw[lc aux, -{Stealth}] (O) -- (1.1,0.75) node[above left, lc label] {$\bm{X}_{:,2}$};

  % Projection X\hat{\theta} and observation y
  \coordinate (Yhat) at (3.6, 0.35);
  \coordinate (Y) at (3.6, 3.15);

  \draw[lc vector, draw=lc-thm, line width=1.1pt] (O) -- (Yhat) node[midway, below right, lc label, text=lc-thm] {$\hat{\bm{y}} = \bm{X}\hat{\bm{\theta}}$};
  \draw[lc vector, draw=lc-velocity, line width=1.1pt] (O) -- (Y) node[midway, above left, lc label, text=lc-velocity] {$\bm{y}$};

  % Residual vector y - X\hat{\theta}
  \draw[lc vector, draw=lc-force, line width=1.0pt] (Yhat) -- (Y) node[midway, right=2pt, lc label, text=lc-force] {$\bm{e} = \bm{y} - \bm{X}\hat{\bm{\theta}}$};

  % Right angle marker at Yhat
  \draw[draw=lc-ink, line width=0.6pt] ($(Yhat)+(-0.26,-0.025)$) -- ($(Yhat)+(-0.26,0.235)$) -- ($(Yhat)+(0,0.26)$);

  % Origin dot
  \fill[lc-ink] (O) circle (1.6pt) node[left=2pt, lc label] {$\bm{0}$};
\end{tikzpicture}
