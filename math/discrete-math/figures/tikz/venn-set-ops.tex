\begin{tikzpicture}[scale=.72, every node/.style={font=\small}]
\def\r{0.72}
\newcommand{\basevenn}[2]{
  \begin{scope}[shift={(#1,0)}]
    \draw[rounded corners] (-1.35,-1.05) rectangle (1.65,1.15);
    \node at (-1.12,.92) {$U$};
    #2
    \draw (-.35,0) circle (\r);
    \draw (.35,0) circle (\r);
    \node at (-.78,.58) {$A$};
    \node at (.78,.58) {$B$};
  \end{scope}
}
\basevenn{0}{
  \begin{scope}
    \clip (-.35,0) circle (\r);
    \fill[blue!28] (.35,0) circle (\r);
  \end{scope}
  \node at (.15,-1.38) {$A\cap B$};
}
\basevenn{3.2}{
  \fill[blue!18] (-.35,0) circle (\r);
  \fill[blue!18] (.35,0) circle (\r);
  \node at (.15,-1.38) {$A\cup B$};
}
\basevenn{6.4}{
  \begin{scope}
    \clip (-.35,0) circle (\r);
    \fill[blue!25,even odd rule] (-.35,0) circle (\r) (.35,0) circle (\r);
  \end{scope}
  \node at (.15,-1.38) {$A-B$};
}
\basevenn{9.6}{
  \fill[blue!22,even odd rule] (-.35,0) circle (\r) (.35,0) circle (\r);
  \node at (.15,-1.38) {$A\oplus B$};
}
\end{tikzpicture}
