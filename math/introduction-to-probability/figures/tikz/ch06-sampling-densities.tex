\begin{tikzpicture}
  % Left panel: Chi-square(n) densities for n = 2, 4, 8
  \begin{axis}[
    name=chi2ax,
    width=6.6cm,
    height=4.8cm,
    lc plot,
    clip=false,
    xmin=0, xmax=16,
    ymin=0, ymax=0.55,
    xtick={0,4,8,12,16},
    ytick={0.2,0.4},
    xlabel={$x$},
    ylabel={$f(x)$},
    every axis x label/.style={at={(ticklabel* cs:1.0)},anchor=west,font=\small},
    every axis y label/.style={at={(ticklabel* cs:1.0)},anchor=south,font=\small},
    samples=120,
    domain=0.05:15.8
  ]
    % n=2: (1/2) exp(-x/2)
    \addplot[lc curve 1,thick] {0.5*exp(-x/2)};
    % n=4: (1/4) x exp(-x/2)
    \addplot[lc curve 2,thick] {0.25*x*exp(-x/2)};
    % n=8: (1/96) x^3 exp(-x/2)
    \addplot[lc curve 3,thick] {(x^3)/96*exp(-x/2)};
    \node[lc label,anchor=west] at (axis cs:1.3,0.44) {$n=2$};
    \node[lc label,anchor=west] at (axis cs:3.6,0.22) {$n=4$};
    \node[lc label,anchor=west] at (axis cs:8.8,0.13) {$n=8$};
    \node[lc label,anchor=north] at (axis cs:8,-0.11) {(1) $\chi^2(n)$ 分布};
  \end{axis}

  % Right panel: t(n) vs N(0,1)
  \begin{axis}[
    at={(chi2ax.outer east)},
    anchor=outer west,
    xshift=0.9cm,
    width=6.6cm,
    height=4.8cm,
    lc plot,
    clip=false,
    xmin=-4.2, xmax=4.2,
    ymin=0, ymax=0.46,
    xtick={-3,0,3},
    ytick={0.2,0.4},
    yticklabel style={fill=white,inner sep=1pt},
    xlabel={$t$},
    ylabel={$h(t)$},
    every axis x label/.style={at={(ticklabel* cs:1.0)},anchor=west,font=\small},
    every axis y label/.style={at={(ticklabel* cs:1.0)},anchor=south,font=\small},
    samples=120,
    domain=-4.0:4.0
  ]
    % N(0,1)
    \addplot[lc curve 1,thick] {1/sqrt(2*pi)*exp(-x^2/2)};
    % t(5): Gamma(3)/(sqrt(5*pi)*Gamma(2.5)) = 2/(sqrt(5*pi)*0.75*sqrt(pi)) = 8/(3*pi*sqrt(5)) = 0.37960669
    \addplot[lc curve 2,thick,dashed] {0.379607*(1+x^2/5)^(-3)};
    % t(1): 1/(pi*(1+x^2))
    \addplot[lc curve 3,thick] {1/(pi*(1+x^2))};
    \node[lc label,anchor=west] at (axis cs:0.85,0.38) {$N(0,1)$};
    \node[lc label,anchor=west] at (axis cs:1.35,0.23) {$t(5)$};
    \node[lc label,anchor=west] at (axis cs:1.95,0.10) {$t(1)$};
    \node[lc label,anchor=north] at (axis cs:0,-0.09) {(2) $t(n)$ 与 $N(0,1)$ 对比};
  \end{axis}
\end{tikzpicture}
